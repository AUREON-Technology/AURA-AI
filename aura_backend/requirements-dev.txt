# Test and development dependencies.
#
# Deliberately separate from requirements.txt: the production image should not
# carry a test runner, and the `aura_core` suite needs nothing from the runtime
# dependency set beyond fastapi/httpx/pydantic, which requirements.txt already
# pins. CI and developers install the hash-locked compile of this file,
# requirements-dev.lock, next to the runtime lock (F-38/SEC-10, GAP-033):
#
#     pip install --require-hashes --extra-index-url https://download.pytorch.org/whl/cpu -r requirements.lock -r requirements-dev.lock
#     pytest
#
# The `aura_core` suite needs no Docker, no network, no API key, and no model —
# see docs/guides/developer-guide.md. That is a design property enforced by the
# fitness tests in tests/architecture/, not a convenience.
#
# The *legacy* suite is a different matter, and the gap cost several CI rounds:
# `python -m unittest discover` imports the whole v1/v2 tree, so it needs the
# runtime dependencies of every module it touches. requirements.lock covers most
# of that, `docker` included now that the engineer agent's sandbox adapter
# (aura_core/adapters/docker_sandbox.py) made it a real runtime dependency —
# testcontainers' own skip check (`docker.from_env().ping()`) rides along with it,
# so the Postgres integration test still skips cleanly rather than erroring.
#
#     pip install --require-hashes --extra-index-url https://download.pytorch.org/whl/cpu -r requirements.lock -r requirements-dev.lock
#     pytest                                   # aura_core
#     TESTING=True python -m unittest discover -p "test_*.py"   # legacy

pytest>=8.0
pytest-asyncio>=0.23
pytest-cov>=5.0

# Runtime image must not carry this (AURA-020). unittest discover in CI installs
# both requirement files, so the Postgres integration test still runs there.
testcontainers[postgres]>=4.0.0

# Lockfile regeneration, in this order (uv; the platform flags make it host-independent).
# Every lock is hashed. --torch-backend cpu sends only the PyTorch packages to PyTorch's
# CPU index and everything else to PyPI alone -- never --extra-index-url with
# --index-strategy unsafe-best-match, which lets either index answer for any name (the
# dependency-confusion shape of the 2022 torchtriton incident). Edit this file or
# requirements.txt -> rerun all three; the dev and training locks are constrained to the
# runtime one, so a shared package can never differ between them.
#   uv pip compile --python-version 3.11 --python-platform linux --strip-extras --torch-backend cpu --generate-hashes --emit-index-annotation --output-file requirements.lock requirements.txt
#   uv pip compile --python-version 3.11 --python-platform linux --strip-extras --torch-backend cpu --generate-hashes --emit-index-annotation --constraint requirements.lock --output-file requirements-dev.lock requirements-dev.txt
#   uv pip compile --python-version 3.11 --python-platform linux --strip-extras --torch-backend cpu --generate-hashes --emit-index-annotation --output-file requirements-train.lock requirements-train.txt
pip-tools>=7.4
