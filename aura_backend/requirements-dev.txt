# Test and development dependencies.
#
# Deliberately separate from requirements.txt: the production image should not
# carry a test runner, and the `aura_core` suite needs nothing from the runtime
# dependency set beyond fastapi/httpx/pydantic, which requirements.txt already
# pins. Install both to run the full suite:
#
#     pip install -r requirements.txt -r requirements-dev.txt
#     pytest
#
# The `aura_core` suite needs no Docker, no network, no API key, and no model —
# see docs/guides/developer-guide.md. That is a design property enforced by the
# fitness tests in tests/architecture/, not a convenience.
#
# The *legacy* suite is a different matter, and the gap cost several CI rounds:
# `python -m unittest discover` imports the whole v1/v2 tree, so it needs the
# runtime dependencies of every module it touches. requirements.txt covers most,
# but `docker` (the client library testcontainers uses to decide whether to skip)
# was implicit, and without it the Postgres integration test errors instead of
# skipping. Pinning it here means a fresh checkout reproduces what CI runs.
#
#     pip install -r requirements.txt -r requirements-dev.txt
#     pytest                                   # aura_core
#     TESTING=True python -m unittest discover -p "test_*.py"   # legacy

pytest>=8.0
pytest-asyncio>=0.23
pytest-cov>=5.0

# testcontainers' skip check is `docker.from_env().ping()`. Absent, the import
# fails and the integration test reports an error rather than skipping cleanly —
# which is the difference between "no Docker here" and "this test is broken".
docker>=7.0
