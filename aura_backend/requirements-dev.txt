# Test and development dependencies.
#
# Deliberately separate from requirements.txt: the production image should not
# carry a test runner, and the `aura_core` suite needs nothing from the runtime
# dependency set beyond fastapi/httpx/pydantic, which requirements.txt already
# pins. Install the lockfile plus this file to run the full suite:
#
#     pip install -r requirements.lock -r requirements-dev.txt
#     pytest
#
# The `aura_core` suite needs no Docker, no network, no API key, and no model —
# see docs/guides/developer-guide.md. That is a design property enforced by the
# fitness tests in tests/architecture/, not a convenience.
#
# The *legacy* suite is a different matter, and the gap cost several CI rounds:
# `python -m unittest discover` imports the whole v1/v2 tree, so it needs the
# runtime dependencies of every module it touches. requirements.lock covers most
# of that, `docker` included now that the engineer agent's sandbox adapter
# (aura_core/adapters/docker_sandbox.py) made it a real runtime dependency —
# testcontainers' own skip check (`docker.from_env().ping()`) rides along with it,
# so the Postgres integration test still skips cleanly rather than erroring.
#
#     pip install -r requirements.lock -r requirements-dev.txt
#     pytest                                   # aura_core
#     TESTING=True python -m unittest discover -p "test_*.py"   # legacy

pytest>=8.0
pytest-asyncio>=0.23
pytest-cov>=5.0

# Runtime image must not carry this (AURA-020). unittest discover in CI installs
# both requirement files, so the Postgres integration test still runs there.
testcontainers[postgres]>=4.0.0

# Lockfile regeneration (runtime pins only; run on Linux 3.11 or with uv's platform flags):
#   uv pip compile --python-version 3.11 --python-platform linux --strip-extras -o requirements.lock requirements.txt
pip-tools>=7.4
