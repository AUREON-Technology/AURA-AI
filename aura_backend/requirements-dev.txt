# Test and development dependencies.
#
# Deliberately separate from requirements.txt: the production image should not
# carry a test runner, and the `aura_core` suite needs nothing from the runtime
# dependency set beyond fastapi/httpx/pydantic, which requirements.txt already
# pins. Install the lockfile plus this file to run the full suite -- as two commands: the lock
# is hash-pinned (audit F-38) and this file is not, and pip will not mix the two in one install:
#
#     pip install --require-hashes --extra-index-url https://download.pytorch.org/whl/cpu -r requirements.lock
#     pip install -r requirements-dev.txt
#     pytest
#
# The `aura_core` suite needs no Docker, no network, no API key, and no model —
# see docs/guides/developer-guide.md. That is a design property enforced by the
# fitness tests in tests/architecture/, not a convenience.
#
# The *legacy* suite is a different matter, and the gap cost several CI rounds:
# `python -m unittest discover` imports the whole v1/v2 tree, so it needs the
# runtime dependencies of every module it touches. requirements.lock covers most
# of that, `docker` included now that the engineer agent's sandbox adapter
# (aura_core/adapters/docker_sandbox.py) made it a real runtime dependency —
# testcontainers' own skip check (`docker.from_env().ping()`) rides along with it,
# so the Postgres integration test still skips cleanly rather than erroring.
#
#     pip install --require-hashes --extra-index-url https://download.pytorch.org/whl/cpu -r requirements.lock
#     pip install -r requirements-dev.txt
#     pytest                                   # aura_core
#     TESTING=True python -m unittest discover -p "test_*.py"   # legacy

pytest>=8.0
pytest-asyncio>=0.23
pytest-cov>=5.0

# Runtime image must not carry this (AURA-020). unittest discover in CI installs
# both requirement files, so the Postgres integration test still runs there.
testcontainers[postgres]>=4.0.0

# This file is the *input*; CI installs requirements-dev.lock, its hash-pinned compilation
# (audit G-06 / GAP-033), constrained by requirements.lock so a package in both carries one
# version. After editing this file, regenerate the lock and commit both:
#   uv pip compile --python-version 3.11 --python-platform linux --strip-extras --generate-hashes -c requirements.lock --output-file requirements-dev.lock requirements-dev.txt
# Install it the same way CI does:
#   pip install --require-hashes -r requirements-dev.lock
#
# Lockfile regeneration (runtime pins only; run on Linux 3.11 or with uv's platform flags).
# --torch-backend cpu sends only the PyTorch packages to PyTorch's CPU index and everything else
# to PyPI (no --extra-index-url / unsafe-best-match, which let either index serve any name), and
# --generate-hashes records a sha256 for every file (audit F-38). Then regenerate
# requirements-train.lock with the command in requirements-train.txt.
#   uv pip compile --python-version 3.11 --python-platform linux --strip-extras --torch-backend cpu --generate-hashes --output-file requirements.lock requirements.txt
